clc;
clear;
% ---------------------- Input Parameters ----------------------
F = 100;                        % Feed flow rate (kmol/h)
z = [0.5; 0.3; 0.2];            % Feed composition (mole fractions)
K = [2.0; 1.0; 0.5];            % Equilibrium constants (y = Kx)
D = 40;                         % Distillate flow
R = 1.5 * D;                    % Reflux flow
B = F - D;                      % Bottoms flow
n = length(z);                  % Number of components

% -------------------- Matrix Initialization --------------------
A = zeros(2*n, 3*n)            % Coefficients matrix
b = zeros(2*n, 1)             % Right-hand side vector

% ------------------ Equation Construction ------------------
for i = 1:n
    % Equation 1: Overall component balance
    % F*z = D*K*x1 + B*xB
    A(i, i) = D * K(i);                  % x1(i)
    A(i, 2*n + i) = B;                   % xB(i)
    b(i) = F * z(i);

    % Equation 2: Intermediate stage component balance
    % F*z + R*x1 = B*xB + (R+D)*x2
    A(n+i, i) = R;                       % x1(i)
    A(n+i, n + i) = -(R + D);            % x2(i)
    A(n+i, 2*n + i) = B;                 % xB(i)
    b(n+i) = -F * z(i);
end

% ------------------ Solve the System ------------------
x = A \ b;   % Solve Ax = b

% ------------------ Extract and Normalize ------------------
x1 = x(1:n);               % Liquid in rectifying section
x2 = x(n+1:2*n);           % Liquid in stripping section
xB = x(2*n+1:end);         % Bottom product
y1 = K .* x1;              % Vapor from rectifying section
y2 = K .* x2;              % Vapor from stripping section

% Normalize to ensure mole fractions add to 1
x1 = x1 / sum(x1);
x2 = x2 / sum(x2);
xB = xB / sum(xB);
y1 = y1 / sum(y1);
y2 = y2 / sum(y2);

% ------------------ Display Output ------------------
fprintf('--- Multicomponent Partial Reflux Distillation Results ---\n\n');
for i = 1:n
    fprintf('Component %d:\n', i);
    fprintf('  x1 (rectifying)   = %.4f\n', x1(i))
    fprintf('  y1 (top vapor)    = %.4f\n', y1(i))
    fprintf('  x2 (stripping)    = %.4f\n', x2(i))
    fprintf('  y2 (mid vapor)    = %.4f\n', y2(i))
    fprintf('  xB (bottoms)      = %.4f\n\n', xB(i))
end
